   176→# ─────────────────────────────────────────────────────────────────────────
   177→# TocDocTemplate with running header + footer (page numbers)
   178→# ─────────────────────────────────────────────────────────────────────────
   179→class TocDocTemplate(SimpleDocTemplate):
   180→    def afterFlowable(self, flowable):
   181→        if hasattr(flowable, 'bookmark_name'):
   182→            level = getattr(flowable, 'bookmark_level', 0)
   183→            text  = getattr(flowable, 'bookmark_text', '')
   184→            key   = getattr(flowable, 'bookmark_key', '')
   185→            self.notify('TOCEntry', (level, text, self.page, key))
   186→
   187→def add_heading(text, style, level=0):
   188→    key = 'h_' + hashlib.md5(text.encode('utf-8')).hexdigest()[:8]
   189→    p = Paragraph('<a name="%s"/>%s' % (key, text), style)
   190→    p.bookmark_name = key
   191→    p.bookmark_level = level
   192→    p.bookmark_text = text
   193→    p.bookmark_key = key
   194→    return p
   195→
   196→def h1(text):
   197→    """Section heading + accent rule + thin gold underline."""
   198→    elems = [
   199→        CondPageBreak(80),
   200→        add_heading(text, style_h1, level=0),
   201→        HRFlowable(width='100%', thickness=1.2, color=ACCENT_3,
   202→                   spaceBefore=0, spaceAfter=2),
   203→        HRFlowable(width='40%', thickness=0.8, color=ACCENT_2,
   204→                   spaceBefore=0, spaceAfter=8, hAlign='LEFT'),
   205→    ]
   206→    return elems
   207→
   208→def h2(text):
   209→    return [add_heading(text, style_h2, level=1)]
   210→
   211→def h3(text):
   212→    return [Paragraph('<b>%s</b>' % text, style_h3)]
   213→
   214→def para(text):
   215→    return Paragraph(text, style_body)
   216→
   217→def para_nf(text):
   218→    return Paragraph(text, style_body_nofirst)
   219→
   220→def bullets(items):
   221→    items_flow = [ListItem(Paragraph(t, style_bullet), leftIndent=12,
   222→                           value='bullet', bulletColor=ACCENT)
   223→                  for t in items]
   224→    return ListFlowable(items_flow, bulletType='bullet', start='\u2022',
   225→                        leftIndent=18, bulletFontSize=9, bulletColor=ACCENT)

   235→        ('TOPPADDING', (0,0), (-1,-1), 6),
   236→        ('BOTTOMPADDING', (0,0), (-1,-1), 6),
   237→        ('BACKGROUND', (0,0), (-1,-1), SECTION_BG),
   238→    ]))

   241→def callout_box(big_text, label_text, width=160):
   242→    inner = Table(
   243→        [[Paragraph('<b>%s</b>' % big_text, style_callout_big)],
   244→         [Paragraph(label_text, style_callout_label)]],
   245→        colWidths=[width]

   247→    inner.setStyle(TableStyle([
   248→        ('BACKGROUND', (0,0), (-1,-1), CARD_BG),
   249→        ('BOX', (0,0), (-1,-1), 1, ACCENT),
   250→        ('LINEBEFORE', (0,0), (0,-1), 4, ACCENT),
   251→        ('TOPPADDING', (0,0), (-1,-1), 8),
   252→        ('BOTTOMPADDING', (0,0), (-1,-1), 8),
   253→        ('LEFTPADDING', (0,0), (-1,-1), 6),
   254→        ('RIGHTPADDING', (0,0), (-1,-1), 6),
   255→        ('VALIGN', (0,0), (-1,-1), 'MIDDLE'),
   256→    ]))
   257→    return inner

   259→def metrics_row(metrics):
   260→    """metrics = list of (big, label) tuples. Returns a single centered Table
   261→    where each column is a callout cell with stacked big + label paragraphs.
   262→    Single-table design (no nested Tables) so QA detects one centered table."""
   263→    n = len(metrics)
   264→    col_w = CONTENT_W / n
   265→    big_row = [Paragraph('<b>%s</b>' % b, style_callout_big) for (b, l) in metrics]
   266→    lbl_row = [Paragraph(l, style_callout_label) for (b, l) in metrics]
   267→    t = Table([big_row, lbl_row], colWidths=[col_w]*n, hAlign='CENTER')
   268→    ts = [
   269→        ('BACKGROUND', (0,0), (-1,-1), CARD_BG),
   270→        ('BOX', (0,0), (-1,-1), 1, ACCENT),
   271→        ('VALIGN', (0,0), (-1,-1), 'MIDDLE'),
   272→        ('ALIGN', (0,0), (-1,-1), 'CENTER'),
   273→        ('TOPPADDING', (0,0), (-1,0), 8),
   274→        ('BOTTOMPADDING', (0,0), (-1,0), 2),
   275→        ('TOPPADDING', (0,1), (-1,1), 0),
   276→        ('BOTTOMPADDING', (0,1), (-1,1), 8),
   277→        ('LEFTPADDING', (0,0), (-1,-1), 4),
   278→        ('RIGHTPADDING', (0,0), (-1,-1), 4),
   279→    ]
   280→    # Vertical accent bars between columns (and on left of first / right of last)
   281→    for c in range(n):
   282→        ts.append(('LINEBEFORE', (c,0), (c,-1), 3 if c == 0 else 0.5, ACCENT))
   283→    ts.append(('LINEAFTER', (n-1,0), (n-1,-1), 3, ACCENT))
   284→    t.setStyle(TableStyle(ts))
   285→    return t

   287→def std_table(header, rows, col_ratios=None, header_bg=HEADER_FILL):
   288→    if col_ratios is None:
   289→        col_ratios = [1.0/len(header)] * len(header)
   290→    col_widths = [r * CONTENT_W for r in col_ratios]
   291→    data = [[Paragraph('<b>%s</b>' % h, style_thead) for h in header]]
   292→    for row in rows:
   293→        data.append([Paragraph(c, style_tcell) if not isinstance(c, Paragraph) else c
   294→                     for c in row])
   295→    t = Table(data, colWidths=col_widths, hAlign='CENTER', repeatRows=1)
   296→    ts = [
   297→        ('BACKGROUND', (0,0), (-1,0), header_bg),
   298→        ('TEXTCOLOR',  (0,0), (-1,0), colors.white),
   299→        ('GRID', (0,0), (-1,-1), 0.5, BORDER),
   300→        ('VALIGN', (0,0), (-1,-1), 'MIDDLE'),
   301→        ('LEFTPADDING', (0,0), (-1,-1), 6),
   302→        ('RIGHTPADDING', (0,0), (-1,-1), 6),
   303→        ('TOPPADDING', (0,0), (-1,-1), 6),
   304→        ('BOTTOMPADDING', (0,0), (-1,-1), 6),
   305→    ]
   306→    for i in range(1, len(data)):
   307→        if i % 2 == 1:
   308→            ts.append(('BACKGROUND', (0,i), (-1,i), colors.white))
   309→        else:
   310→            ts.append(('BACKGROUND', (0,i), (-1,i), TABLE_STRIPE))
   311→    t.setStyle(TableStyle(ts))
   312→    return t
