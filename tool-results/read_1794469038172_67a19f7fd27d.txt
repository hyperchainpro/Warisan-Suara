   164→function formatSpeakers(n: number): string {
   165→  if (n >= 1_000_000) {
   166→    const m = n / 1_000_000;
   167→    // 38M, 2.7M
   168→    return `${m % 1 === 0 ? m.toFixed(0) : m.toFixed(1)}M`;
   169→  }
   170→  if (n >= 1_000) {
   171→    const k = n / 1_000;
   172→    return `${k % 1 === 0 ? k.toFixed(0) : k.toFixed(0)}K`;
   173→  }
   174→  return String(n);
   175→}

   670→export function LanguageMapSection() {
   671→  const [languages, setLanguages] = useState<Language[]>([]);
   672→  const [loading, setLoading] = useState(true);
   673→  const [error, setError] = useState<string | null>(null);
   674→  const [filter, setFilter] = useState<FilterKey>("ALL");
   675→  const [selected, setSelected] = useState<Language | null>(null);
   676→  const [dialogOpen, setDialogOpen] = useState(false);
   677→  const [traditions, setTraditions] = useState<Tradition[]>([]);
   678→  const [loadingTraditions, setLoadingTraditions] = useState(false);
   679→  const [traditionsError, setTraditionsError] = useState<string | null>(null);
   680→  // Bumping this nonce re-triggers the languages fetch (e.g. on retry).
   681→  const [fetchNonce, setFetchNonce] = useState(0);
   682→
   683→  // Fetch languages whenever the nonce changes.
   684→  // setState calls happen only inside async callbacks to avoid cascading
   685→  // renders (react-hooks/set-state-in-effect). Initial loading=true covers
   686→  // the first render before data arrives.
   687→  useEffect(() => {
   688→    const controller = new AbortController();
   689→    fetch("/api/languages", { signal: controller.signal })
   690→      .then((r) => {
   691→        if (!r.ok) throw new Error("Gagal memuat daftar bahasa");
   692→        return r.json();
   693→      })
   694→      .then((data: { languages: Language[] }) => {
   695→        setLanguages(data.languages ?? []);
   696→        setError(null);
   697→        setLoading(false);
   698→      })
   699→      .catch((err) => {
   700→        if (controller.signal.aborted) return;
   701→        setError(err?.message ?? "Gagal memuat daftar bahasa");
   702→        setLoading(false);
   703→      });
   704→    return () => controller.abort();
   705→  }, [fetchNonce]);
   706→
   707→  // Open a language detail dialog and start fetching its traditions.
   708→  // Runs in an event-handler context, so synchronous setState is allowed.
   709→  const handleOpen = useCallback((lang: Language) => {
   710→    setSelected(lang);
   711→    setDialogOpen(true);
   712→    setTraditions([]);
   713→    setTraditionsError(null);
   714→    setLoadingTraditions(true);
   715→    fetch(`/api/traditions?languageCode=${encodeURIComponent(lang.code)}`)
   716→      .then((r) => {
   717→        if (!r.ok) throw new Error("Gagal memuat tradisi");
   718→        return r.json();
   719→      })
   720→      .then((data: { traditions: Tradition[] }) => {
   721→        setTraditions(data.traditions ?? []);
   722→        setLoadingTraditions(false);
   723→      })
   724→      .catch((err) => {
   725→        setTraditionsError(err?.message ?? "Gagal memuat tradisi");
   726→        setLoadingTraditions(false);
   727→      });
   728→  }, []);
   729→
   730→  // Derived stats
   731→  const stats = useMemo(() => {
   732→    const totalSpeakers = languages.reduce((sum, l) => sum + l.speakers, 0);
   733→    const endangeredCount = languages.filter(
   734→      (l) =>
   735→        l.vitality === "TERANCAM" ||
   736→        l.vitality === "KRITIS" ||
   737→        l.vitality === "HAMPIR_PUNAH"
   738→    ).length;
   739→    return {
   740→      total: languages.length,
   741→      totalSpeakers,
   742→      endangered: endangeredCount,
   743→    };
   744→  }, [languages]);
   745→
   746→  // Filtered languages
   747→  const filtered = useMemo(() => {
   748→    if (filter === "ALL") return languages;
   749→    return languages.filter((l) => l.vitality === filter);
   750→  }, [languages, filter]);
   751→
   752→  return (
   753→    <section
   754→      id="peta-bahasa"
   755→      className="relative py-16 sm:py-20 border-b border-border overflow-hidden"
   756→    >
   757→      {/* Decorative batik band at top */}
   758→      <BatikDivider />
   759→
   760→      {/* Soft background glow */}
   761→      <div
   762→        aria-hidden="true"
   763→        className="pointer-events-none absolute inset-0 overflow-hidden"
   764→      >
   765→        <div className="absolute -top-20 left-1/4 h-72 w-72 rounded-full bg-primary/5 blur-3xl" />
   766→        <div className="absolute top-32 -right-24 h-80 w-80 rounded-full bg-accent/10 blur-3xl" />
   767→        <div className="absolute bottom-0 left-1/3 h-64 w-64 rounded-full bg-destructive/5 blur-3xl" />
   768→      </div>
   769→
   770→      <div className="relative container mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   771→        {/* Header */}
   772→        <div className="max-w-3xl space-y-3">
   773→          <div className="inline-flex items-center gap-2">
   774→            <span className="inline-flex items-center gap-1.5 rounded-full border border-primary/30 bg-primary/10 px-3 py-1 text-xs font-medium text-primary">
   775→              <MapPin className="h-3.5 w-3.5" />
   776→              Peta Bahasa
   777→            </span>
   778→            {!loading && (
   779→              <Badge variant="secondary" className="bg-accent/30 border-accent/40">
   780→                {stats.total} bahasa terdata
   781→              </Badge>

   785→          <h2 className="text-3xl sm:text-4xl font-bold tracking-tight text-foreground">
